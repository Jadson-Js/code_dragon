// ==========================================
// CORE TABLES
// ==========================================

Table users {
  id integer [primary key]
  name varchar [not null]
  email varchar [not null, unique]
  password varchar [not null]
  birth_date date
  verified_at timestamp
  image_id integer
  linkedin_url varchar
  github_url varchar
  portfolio_url varchar
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  deleted_at timestamp
  
  indexes {
    email [unique]
    verified_at
    deleted_at
  }
}

Table email_tokens {
  id integer [primary key]
  user_id integer [not null]
  token varchar [not null, unique]
  type enum('verification', 'password_reset', 'email_change') [not null]
  used_at timestamp
  expires_at timestamp [not null]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  
  indexes {
    token [unique]
    user_id
    expires_at
    (user_id, type, used_at)
  }
}

Table images {
  id integer [primary key]
  url varchar [not null]
  key varchar [not null]
  mime_type enum('image/jpeg', 'image/jpg', 'image/png', 'image/webp') [not null]
  size integer [not null]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  
  indexes {
    key [unique]
  }
}

  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  deleted_at timestamp

    subject_id
    (seniority_id, speciality_id, subject_id)
    (quiz_objective_id, seniority_id, speciality_id)  // ← ADICIONAR
    deleted_at
    reports
  }
  
  Note: 'Quiz question bank - alternatives stored as JSON array'
}
